public class Trilal {
    public static void main(String[] args) {
        for (int i = 9; i >= 0; i--) {
            for (int j = 0; j < i*2; j++) {
                System.out.print(".");
            }
            for (int j = i; j <= 9; j++) {
                System.out.print(".");
            }
            for (int j = i; j <= 9; j++) {
                System.out.print(j);
            }
            for (int j = 8; j >= i; j--) {
                System.out.print(j);
            }
            for (int j = 0; j < i*2; j++) {
                System.out.print(".");
            }
            for (int j = 9; j >= i; j--) {
                System.out.print(".");
            }
            System.out.println();
        }
        for (int i = 9; i >= 0 ; i--) {
            for (int j = 0; j < 2; j++) {
                for (int k = 0; k < i; k++) {
                    System.out.print(".");
                }
                for (int k = i; k <= 9; k++) {
                    System.out.print(k);
                }
                for (int k = 8; k >= i; k--) {
                    System.out.print(k);
                }
                for (int k = i; k >= 0; k--) {
                    if (j == 0 && k >= 0) {
                        System.out.print(".");
                    }
                    if (j == 1 && k > 0) {
                        System.out.print(".");
                    }
                }
            }
            System.out.println();
        }
    }
}
