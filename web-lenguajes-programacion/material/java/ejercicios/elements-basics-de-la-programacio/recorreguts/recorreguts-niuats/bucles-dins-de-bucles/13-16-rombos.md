public class Rombos {
    public static void main(String[] args) {
        System.out.println("Quants?");
        int quantitat = Integer.parseInt(Entrada.readLine());

        for (int i = 0; i < quantitat; i++) {
            for (int j = 9; j >= 0 ; j--) {
                for (int k = 0; k < j; k++) {
                    System.out.print(".");
                }
                for (int k = j; k <= 9; k++) {
                    System.out.print(k);
                }
                for (int k = 8; k >= j; k--) {
                    System.out.print(k);
                }
                for (int k = j; k > 0; k--) {
                    System.out.print(".");
                }
                System.out.println();
            }
            for (int j = 1; j <= 9 ; j++) {
                for (int k = 0; k < j; k++) {
                    System.out.print(".");
                }
                for (int k = j; k <= 9; k++) {
                    System.out.print(k);
                }
                for (int k = 8; k >= j; k--) {
                    System.out.print(k);
                }
                for (int k = 0; k < j; k++) {
                    System.out.print(".");
                }
                System.out.println();
            }
        }
    }
}
